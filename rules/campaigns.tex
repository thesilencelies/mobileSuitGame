% Absolute-coordinate standee (used by board setup and \tilestandee)
\newcommand{\standee}[4]{% x, y, art, ring colour
  \fill[white] (#1,#2) circle (\standeeradius);
  \fill[#4!22!white] (#1,#2) circle (\standeeradius);
  \begin{scope}
    \clip (#1,#2) circle (\standeeclipradius);
    \node at (#1,#2+\standeeimgyoffset) {\includegraphics[height=\standeeimgsize cm]{#3}};
  \end{scope}
  \draw[line width=3pt, #4] (#1,#2) circle (\standeeradius);
}

% Tile-coordinate standee: #1=col (0..2), #2=row (0..3), #3=art, #4=ring colour

% Lay out the four cards: #1 #2 = Ripper's slots, #3 #4 = Kuwagata's slots.
% Each card carries a \maxheight background image that recomputes to the wrong
% size if tikz re-typesets the node content during its layout passes, shifting
% the card's contents out of place. So we typeset each slot into a save box in
% normal outer context first (where \maxheight resolves correctly), then drop
% the finished boxes into the nodes with \usebox -- no re-typesetting, no shift.
\newsavebox{\slotAbox}\newsavebox{\slotBbox}\newsavebox{\slotCbox}\newsavebox{\slotDbox}

    \node[slot] (a) at (0,0) {\usebox{\slotAbox}};
    \node[slot, right=0.22cm of a] (b) {\usebox{\slotBbox}};
    \node[slot, right=0.7cm of b] (cc) {\usebox{\slotCbox}};
    \node[slot, right=0.22cm of cc] (d) {\usebox{\slotDbox}};

% Wrap a terrain card so standees / arrows can be drawn on it using tile
% coordinates (c, r). #1 is the terrain card .tex. A terrain tikzpicture uses
% a \maxheight background image, which is shifted out of place if the card is
% \input directly inside a tikz node. So we first typeset the card into a save box
% (normal context, where \maxheight resolves correctly) and then place that finished
% box in the node. Named coordinates (tile-\c-\r) are defined for each tile.
\newsavebox{\terraincardbox}
\newenvironment{terrainscene}[1]
  {\sbox{\terraincardbox}{\scalebox{0.8}{\input{#1}}}%
   \begin{tikzpicture}[scale=0.8]
     \node[anchor=south west, inner sep=0pt] (T) at (0,0){\usebox{\terraincardbox}};
     \begin{scope}[shift={(T.south west)}]
       \foreach \c in {0,1,2}{%
         \foreach \r in {0,1,2,3}{%
           \coordinate (tile-\c-\r) at ({\tcHOffset + (\c + 0.5)*\tcTileSize}, {\tcVOffset + (\r + 0.5)*\tcTileSize});
         }%
       }%
  }
  {\end{scope}\end{tikzpicture}}

of the campaign will decide how it will change.
